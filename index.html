async function startScanner() { const status = document.getElementById('status'); const button = document.getElementById('startButton'); button.disabled = true; status.textContent = 'Initializing camera...'; try { if (typeof Html5Qrcode === 'undefined') { throw new Error('QR scanner library did not load.'); } if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) { throw new Error('Camera API is unavailable. Open the HTTPS website in Chrome.'); } if (!scanner) { scanner = new Html5Qrcode('reader'); } await scanner.start( { facingMode: 'environment' }, { fps: 10, qrbox: { width: 250, height: 250 } }, function(decodedText) { if (busy) return; busy = true; status.textContent = 'QR detected: ' + decodedText; scanner.stop().then(function() { openGoogleForm(decodedText); }).catch(function() { openGoogleForm(decodedText); }); }, function(errorMessage) { // Normal: no QR detected in this frame. } ); status.textContent = 'Camera started. Point it at the QR code.'; } catch (error) { status.textContent = 'Camera error: ' + error.message; button.disabled = false; } }
